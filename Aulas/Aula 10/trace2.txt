class Pessoa {
	String nome;
	int idade;

	public Pessoa(String n, int id) {
		this.nome = n;
		this.idade = id;
	}
	
	@Override
	public String toString() {
		String tmp = this.nome + " --> ";
		tmp += idade;
		return tmp;
	}

	public void mostrarId() {
		System.out.println(this);
	}
}

}
